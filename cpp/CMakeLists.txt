# The verification container keeps all executables/libraries but need not retain
# duplicate object files. Only explicit source .o files in THIS build tree
# are removed, after successful linking; normal developer builds are unaffected.
if(MLAT_EPHEMERAL_TEST_OBJECTS)
  if(NOT CMAKE_GENERATOR STREQUAL "Ninja" OR NOT UNIX)
    message(FATAL_ERROR "Ephemeral test objects require the Linux Ninja lab")
  endif()
  get_property(lab_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
  foreach(lab_target IN LISTS lab_targets)
    if(lab_target MATCHES "^mlat_[a-z_]+_tests$" OR lab_target STREQUAL "mlat_core" OR lab_target STREQUAL "mlat_server")
      get_target_property(test_sources ${lab_target} SOURCES)
      foreach(test_source IN LISTS test_sources)
        if(NOT test_source MATCHES "^(tests|src)/[a-z_]+\\.cpp$")
          message(FATAL_ERROR "Unexpected test source; refuse object cleanup")
        endif()
        add_custom_command(TARGET ${lab_target} POST_BUILD
          COMMAND ${CMAKE_COMMAND} -E rm -f
            "${CMAKE_CURRENT_BINARY_DIR}/CMakeFiles/${lab_target}.dir/${test_source}.o"
          VERBATIM)
      endforeach()
    endif()
  endforeach()
endif()
